Constraints Reference

Constraints are rules the database engine enforces automatically: PRIMARY KEY, FOREIGN KEY, UNIQUE, NOT NULL, CHECK, and DEFAULT.

Constraints Reference 

/* ----------------------------------------------------------------------------
  3.  CONSTRAINTS REFERENCE
   ---------------------------------------------------------------------------- */
 
-- (a) PRIMARY KEY — uniquely identifies each row; implies NOT NULL + UNIQUE
--     ALREADY USED ABOVE: schools.school_id, visit_id, region_id, etc. Composite primary key example:

CREATE TABLE school_partner_assignment (
    school_id    VARCHAR(10) NOT NULL,
    partner_name VARCHAR(80) NOT NULL,
    start_date   DATE,
    PRIMARY KEY (school_id, partner_name),                -- composite PK
    CONSTRAINT fk_spa_school FOREIGN KEY (school_id) REFERENCES schools(school_id)
) ENGINE=InnoDB;
 
-- (b) FOREIGN KEY — enforces referential integrity to a parent table ON DELETE / ON UPDATE actions: 
--     CASCADE, SET NULL, RESTRICT, NO ACTION (MySQL's InnoDB has no SET DEFAULT support for FKs, unlike Postgres)

ALTER TABLE monitoring_visits
    DROP FOREIGN KEY fk_visits_school,
    ADD CONSTRAINT fk_visits_school
        FOREIGN KEY (school_id) REFERENCES schools(school_id)
        ON DELETE RESTRICT      -- block deleting a school that still has visit records
        ON UPDATE CASCADE;      -- if a school_id value changes, cascade it here
 
-- (c) NOT NULL — column may never store NULL e.g. schools.school_name NOT NULL  (already declared above)
 
-- (d) UNIQUE — no duplicate values (a single NULL is allowed; MySQL allows MULTIPLE NULLs in a UNIQUE column, same as Postgres)

ALTER TABLE devices ADD UNIQUE KEY uq_device_id (device_id);   -- illustrative (already PK)
 
-- (e) CHECK — arbitrary boolean condition each row must satisfy (MySQL 8.0.16+)

ALTER TABLE monitoring_visits
    ADD CONSTRAINT chk_attendance_le_enrolment
    CHECK (attendance_today <= enrolment_total);
 
-- (f) DEFAULT — value used automatically when none supplied

ALTER TABLE monitoring_visits ALTER COLUMN 
    created_at SET DEFAULT CURRENT_TIMESTAMP;
 
-- (g) MySQL has no EXCLUSION constraint (that is a PostgreSQL-only feature, used e.g. to block overlapping 
-- date ranges via GiST indexes). In MySQL the equivalent has to be enforced in application logic or a trigger.
 
-- (h) Dropping constraints — syntax differs by constraint type in MySQL

ALTER TABLE monitoring_visits 
    DROP CHECK chk_attendance_le_enrolment;   -- CHECK

ALTER TABLE modifies an existing table's structure — adding, renaming, or dropping columns, or changing constraints — without rebuilding it from scratch.

Altering Tables 

/* ----------------------------------------------------------------------------
  4.  ALTERING TABLES
   ---------------------------------------------------------------------------- */
 
-- Add a column
ALTER TABLE schools 
    ADD COLUMN head_teacher_name VARCHAR(100);
 
-- Add a column with a default (backfills existing rows)
ALTER TABLE schools 
    ADD COLUMN feeding_program_tier VARCHAR(10) DEFAULT 'Tier 1';
 
-- Add a column in a specific position (MySQL-only convenience — no equivalent in Postgres)
ALTER TABLE schools 
    ADD COLUMN notes TEXT AFTER head_teacher_name;
 
-- Modify a column's data type (keeps the same column name)
ALTER TABLE monitoring_visits MODIFY COLUMN dropout_reason VARCHAR(200);

-- PostgreSQL: ALTER TABLE monitoring_visits ALTER COLUMN dropout_reason TYPE VARCHAR(200);
 
-- Rename a column (and, if needed, retype it in the same statement)
ALTER TABLE schools 
    CHANGE COLUMN head_teacher_name headteacher_name VARCHAR(100);

-- MySQL 8.0+ also supports the simpler:
ALTER TABLE schools 
    RENAME COLUMN headteacher_name TO head_teacher_name;
 
-- Rename a table
RENAME TABLE schools_archive TO schools_archive_2025;
RENAME TABLE schools_archive_2025 TO schools_archive;    -- renamed back

-- (ALTER TABLE schools_archive RENAME TO schools_archive_2025; also works in MySQL)
-- PostgreSQL: ALTER TABLE schools_archive RENAME TO schools_archive_2025;
 
-- Set / drop NOT NULL — MySQL does this via MODIFY COLUMN, restating the full definition
ALTER TABLE schools MODIFY COLUMN head_teacher_name VARCHAR(100) NOT NULL;
ALTER TABLE schools MODIFY COLUMN head_teacher_name VARCHAR(100) NULL;
 
-- Add / drop a constraint after table creation
ALTER TABLE schools ADD CONSTRAINT chk_tier CHECK (feeding_program_tier IN ('Tier 1','Tier 2','Tier 3'));
ALTER TABLE schools DROP CHECK chk_tier;
 
-- Drop a column
ALTER TABLE schools DROP COLUMN feeding_program_tier;
 
-- Add multiple changes in one statement
ALTER TABLE monitoring_visits
    ADD COLUMN reviewed_by VARCHAR(60),
    ADD COLUMN reviewed_at DATETIME;
 
-- Table/column comments (MySQL supports these inline, not via a separate COMMENT ON statement)
ALTER TABLE monitoring_visits COMMENT = 'One row per enumerator visit to a school in a given month';
ALTER TABLE monitoring_visits MODIFY COLUMN meals_per_student DECIMAL(6,2)
    COMMENT 'meals_served / attendance_today';

-- PostgreSQL: COMMENT ON TABLE monitoring_visits IS '...';  COMMENT ON COLUMN ... IS '...';

Dropping and Truncating

DROP TABLE removes structure and data permanently. TRUNCATE TABLE empties all rows while keeping the structure, and is faster than DELETE since it deallocates pages instead of logging each row.

Dropping and Truncating 

/* ----------------------------------------------------------------------------
  5.  DROPPING AND TRUNCATING
   ---------------------------------------------------------------------------- */
 
-- DELETE (DML, row-by-row, can be filtered, is transaction-logged, slow on big tables) covered in Section 2.3
 
-- TRUNCATE — instantly removes ALL rows, resets AUTO_INCREMENT counters, minimal logging, cannot be filtered with WHERE. 
-- In MySQL/InnoDB, TRUNCATE implicitly commits (it is NOT rolled back by a surrounding transaction) — this differs from 
-- PostgreSQL, where TRUNCATE IS transactional.

TRUNCATE TABLE tmp_low_stock_schools;
 
-- Disable FK checks temporarily if you need to TRUNCATE a table that is referenced by others (use with care, and re-enable immediately after)
SET FOREIGN_KEY_CHECKS = 0;

-- TRUNCATE TABLE schools;
SET FOREIGN_KEY_CHECKS = 1;

-- PostgreSQL: TRUNCATE TABLE schools RESTART IDENTITY CASCADE;  -- CASCADE handles dependent tables directly
 
-- DROP TABLE — removes the table structure and all its data permanently
DROP TABLE IF EXISTS schools_archive;
 
-- DROP COLUMN vs DROP TABLE vs DROP DATABASE — increasing scope of destruction
DROP DATABASE school_feeding_staging;   -- drops everything inside it
 
-- Safe pattern: rename-then-drop, so you can recover if something goes wrong
RENAME TABLE complaint_types_lookup TO complaint_types_lookup_old;
DROP TABLE complaint_types_lookup_old;

Indexes and Views

An index speeds up lookups on chosen columns at the cost of extra storage. A view is a stored, named query that behaves like a virtual table.

Indexes and Views 

/* ----------------------------------------------------------------------------
  6.  INDEXES AND VIEWS
   ---------------------------------------------------------------------------- */
 
-- INDEXES speed up lookups/joins/sorts at the cost of extra storage & slower writes
-- Note: InnoDB DOES auto-index foreign key columns in MySQL (unlike Postgres,
-- where you must always index FK columns yourself) — but explicit, well-named
-- indexes are still good practice for composite/query-specific access paths.
 
-- Single-column index
CREATE INDEX idx_visits_school_id ON monitoring_visits (school_id);
 
-- Composite (multi-column) index — column order matters; put the most
-- selective / most-filtered-on column first
CREATE INDEX idx_visits_month_school ON monitoring_visits (report_month, school_id);
 
-- Unique index (equivalent to a UNIQUE constraint, but created separately)
CREATE UNIQUE INDEX uq_idx_school_month ON monitoring_visits (school_id, report_month);
 
-- Functional / expression index (MySQL 8.0.13+) — index on LOWER(school_name)
CREATE INDEX idx_schools_name_lower ON schools ((LOWER(school_name)));
-- PostgreSQL: CREATE INDEX idx_schools_name_lower ON schools (LOWER(school_name));
 
-- Full-text index — MySQL-native alternative for text search
CREATE FULLTEXT INDEX ft_visits_stock_reason ON monitoring_visits (stock_reason);
SELECT * FROM monitoring_visits
WHERE  MATCH(stock_reason) AGAINST ('late_delivery' IN NATURAL LANGUAGE MODE);
 
-- MySQL has NO native partial/filtered index (e.g. "WHERE stock_out = 'yes'").
-- PostgreSQL: CREATE INDEX idx_visits_stockouts ON monitoring_visits (school_id) WHERE stock_out = 'yes';
-- Closest MySQL workaround: a generated column flagging the condition, then index that:
ALTER TABLE monitoring_visits
    ADD COLUMN is_stockout TINYINT(1) GENERATED ALWAYS AS (stock_out = 'yes') STORED;
CREATE INDEX idx_visits_is_stockout ON monitoring_visits (is_stockout);
 
-- View existing indexes on a table
SHOW INDEX FROM monitoring_visits;
-- PostgreSQL: SELECT indexname, indexdef FROM pg_indexes WHERE tablename = 'monitoring_visits';
 
-- Drop an index
DROP INDEX idx_schools_name_lower ON schools;
-- PostgreSQL: DROP INDEX IF EXISTS idx_schools_name_lower;
 
-- VIEWS — saved, reusable SELECT statements; do not store data themselves
 
-- Simple view: one row per school with its full location hierarchy
CREATE VIEW v_school_directory AS
SELECT  s.school_id, s.school_name, s.school_type, s.ownership,
        sc.sub_county_name, c.county_name, r.region_name
FROM    schools s
JOIN    sub_counties sc ON sc.sub_county_id = s.sub_county_id
JOIN    counties     c  ON c.county_id      = sc.county_id
JOIN    regions       r ON r.region_id       = c.region_id;
 
SELECT * FROM v_school_directory WHERE region_name = 'Coast';
 
-- Aggregated view: monthly performance per school
CREATE VIEW v_school_monthly_performance AS
SELECT  school_id,
        report_month,
        attendance_rate,
        dropout_rate,
        meals_served,
        ROUND(meals_served / NULLIF(attendance_today,0), 2) AS meals_per_attendee
FROM    monitoring_visits;
 
-- Updatable view (single-table, no aggregation) — writes pass through to the base table
CREATE VIEW v_public_schools AS
SELECT * FROM schools WHERE ownership = 'Public';
 
UPDATE v_public_schools SET head_teacher_name = 'Jane Wanjiru' WHERE school_id = 'SCH0001';
 
-- MySQL has NO native MATERIALIZED VIEW. The standard workaround is a real
-- table that you refresh on a schedule (via the built-in Event Scheduler
-- or an external job):
CREATE TABLE mv_region_summary (
    region_name         VARCHAR(50) PRIMARY KEY,
    schools_covered      INT,
    avg_attendance_rate    DECIMAL(5,1),
    total_meals_served       BIGINT
) ENGINE=InnoDB;
 
-- PostgreSQL native equivalent:
-- CREATE MATERIALIZED VIEW mv_region_summary AS SELECT ... ;
-- REFRESH MATERIALIZED VIEW CONCURRENTLY mv_region_summary;
 
DELIMITER $$
CREATE EVENT ev_refresh_region_summary
ON SCHEDULE EVERY 1 DAY
DO
BEGIN
    TRUNCATE TABLE mv_region_summary;
    INSERT INTO mv_region_summary (region_name, schools_covered, avg_attendance_rate, total_meals_served)
    SELECT r.region_name,
           COUNT(DISTINCT s.school_id),
           ROUND(AVG(mv.attendance_rate), 1),
           SUM(mv.meals_served)
    FROM   monitoring_visits mv
    JOIN   schools s ON s.school_id = mv.school_id
    JOIN   sub_counties sub ON sub.sub_county_id = s.sub_county_id
    JOIN   counties c ON c.county_id = sub.county_id
    JOIN   regions r ON r.region_id = c.region_id
    GROUP BY r.region_name;
END$$
DELIMITER ;
-- SET GLOBAL event_scheduler = ON;   -- required once for scheduled events to run
 
-- Drop a view
DROP VIEW IF EXISTS v_public_schools;

Updating Data

UPDATE modifies existing rows matching a condition. Omitting WHERE updates every row in the table.

Updating Data 

/* ----------------------------------------------------------------------------
  2.  UPDATING DATA
---------------------------------------------------------------------------- */
 
-- Basic single-row update
UPDATE schools
SET    head_teacher_name = 'Jane Wanjiru'
WHERE  school_id = 'SCH0001';
 
-- Update multiple columns at once
UPDATE monitoring_visits
SET    reviewed_by = 'Maurice Juma',
       reviewed_at = NOW()
WHERE  visit_id = 1;
 
-- Conditional / computed update across many rows
UPDATE monitoring_visits
SET    meals_per_student = ROUND(meals_served / NULLIF(attendance_today, 0), 2)
WHERE  meals_served > 0;
 
-- Update using a join to another table (MySQL multi-table UPDATE syntax)
UPDATE monitoring_visits mv
JOIN   schools s ON mv.school_id = s.school_id
SET    mv.partner_perf = 'unrated'
WHERE  s.ownership = 'Private'
  AND  mv.partner_perf IS NULL;

-- UPDATE monitoring_visits mv
-- SET    partner_perf = 'unrated'

-- Update with a subquery in the SET clause
UPDATE schools
SET    feeding_program_tier =
       (SELECT CASE WHEN AVG(mv.attendance_rate) >= 90 THEN 'Tier 1'
                     WHEN AVG(mv.attendance_rate) >= 75 THEN 'Tier 2'
                     ELSE 'Tier 3' END
        FROM monitoring_visits mv WHERE mv.school_id = schools.school_id)
WHERE  EXISTS (SELECT 1 FROM monitoring_visits mv WHERE mv.school_id = schools.school_id);

UPDATE monitoring_visits SET stock_out = 'yes' WHERE stock_level = 0;
SELECT ROW_COUNT();               -- number of rows affected by the last statement

Data Query Language

DQL centers on SELECT — retrieving, shaping, and summarizing data without changing it. Clauses combine in the order SELECT, FROM, WHERE, GROUP BY, HAVING, ORDER BY.

Select

SELECT specifies which columns to retrieve. Aliases (AS) improve readability, and DISTINCT removes duplicate rows.

Select 

/* ----------------------------------------------------------------------------
  1.  SELECT — the basics
---------------------------------------------------------------------------- */
 
SELECT * FROM schools;                                    -- all columns
SELECT school_id, school_name, school_type FROM schools LIMIT 5;
 
SELECT school_name AS name, school_type AS type FROM schools;   -- column aliasing
SELECT DISTINCT school_type FROM schools;                        -- unique values only
SELECT DISTINCT region_name, county_name FROM v_school_directory; -- multi-column distinct
 
-- Computed / derived columns
SELECT school_id,
       enrolment_boys + enrolment_girls               AS recomputed_total,
       ROUND(attendance_today * 100.0 / NULLIF(enrolment_total,0), 1) AS live_attendance_pct
FROM   monitoring_visits;

Filtering with WHERE

WHERE narrows results to rows matching a condition, using comparison operators, AND/OR/NOT, and helpers like IN, BETWEEN, LIKE, and IS NULL.

Filtering with WHERE 

/* ----------------------------------------------------------------------------
  2.  FILTERING WITH WHERE
---------------------------------------------------------------------------- */
 
SELECT * FROM monitoring_visits WHERE report_month = '2025-03';
 
SELECT * FROM monitoring_visits
WHERE  attendance_rate < 80 AND stock_out = 'yes';               -- AND
 
SELECT * FROM monitoring_visits
WHERE  meal_type = 'lunch' OR meal_type = 'both';                 -- OR
 
SELECT * FROM schools WHERE school_type <> 'Primary Day';         -- NOT EQUAL (or !=)
 
SELECT * FROM monitoring_visits
WHERE  report_month BETWEEN '2025-01' AND '2025-06';              -- BETWEEN (inclusive)
 
SELECT * FROM schools
WHERE  school_type IN ('Primary Boarding','Secondary Boarding');  -- IN
 
SELECT * FROM monitoring_visits WHERE meal_type IS NULL;          -- IS NULL (never "= NULL")
SELECT * FROM monitoring_visits WHERE meal_type IS NOT NULL;
 
SELECT * FROM schools WHERE school_name LIKE 'Kwale%';            -- LIKE: % = any chars
SELECT * FROM schools WHERE school_name LIKE '%Secondary%';
-- MySQL's default collation (e.g. utf8mb4_0900_ai_ci) is case-INsensitive,
-- so plain LIKE already behaves like ILIKE. For a case-SENSITIVE match:
SELECT * FROM schools WHERE school_name LIKE BINARY 'kwale%';
-- PostgreSQL: WHERE school_name ILIKE 'kwale%';   -- explicit case-insensitive operator
 
SELECT * FROM schools WHERE school_name REGEXP '^Kwale.*Secondary$';   -- regex match
-- PostgreSQL: WHERE school_name ~ '^Kwale.*Secondary$';
 
SELECT * FROM monitoring_visits
WHERE  NOT (stock_out = 'yes' AND ration_full = 'no');            -- NOT + grouped logic
 
SELECT * FROM monitoring_visits
WHERE  (attendance_rate < 70 OR dropout_rate > 5)
  AND  report_month = '2025-04';                                  -- combining AND/OR with parentheses

Sorting and Limiting

ORDER BY sorts results ascending or descending. LIMIT/OFFSET (or TOP, or FETCH FIRST) restrict and paginate the rows returned.

Sorting and Limiting 

/* ----------------------------------------------------------------------------
  3.  SORTING AND LIMITING
---------------------------------------------------------------------------- */
 
SELECT school_id, report_month, attendance_rate
FROM   monitoring_visits
ORDER BY attendance_rate DESC;                                    -- descending
 
SELECT school_id, report_month, attendance_rate, dropout_rate
FROM   monitoring_visits
ORDER BY report_month ASC, attendance_rate DESC;                   -- multi-column sort
 
SELECT school_id, attendance_rate
FROM   monitoring_visits
ORDER BY attendance_rate DESC
LIMIT 10;                                                          -- top 10
 
SELECT school_id, attendance_rate
FROM   monitoring_visits
ORDER BY attendance_rate DESC
LIMIT 10 OFFSET 10;                                                -- page 2 of 10
-- Equivalent MySQL shorthand:  LIMIT 10, 10;   (offset, row_count — easy to mix up, be careful)
 
SELECT school_id, attendance_rate
FROM   monitoring_visits                   -- MySQL has no separate
ORDER BY attendance_rate DESC              -- FETCH FIRST syntax before 8.0;
LIMIT 5;                                   -- LIMIT is the idiomatic form.

-- PostgreSQL / ANSI standard: ... FETCH FIRST 5 ROWS ONLY;
 
-- Controlling where NULLs sort — MySQL has no NULLS LAST/FIRST keyword; use an expression that pushes NULLs to the end manually:

SELECT school_id, stock_out, attendance_rate
FROM   monitoring_visits
ORDER BY (stock_out IS NULL), attendance_rate DESC;
-- PostgreSQL: ORDER BY stock_out NULLS LAST, attendance_rate DESC;   -- native keyword

Aggregate Functions and Grouping

COUNT, SUM, AVG, MIN, MAX collapse rows into summary values. GROUP BY buckets rows before aggregating, and HAVING filters the grouped results.

Aggregate Functions and Grouping 

/* ----------------------------------------------------------------------------
  4.  AGGREGATE FUNCTIONS AND GROUPING
---------------------------------------------------------------------------- */
 
SELECT COUNT(*)                        AS total_visits FROM monitoring_visits;
SELECT COUNT(DISTINCT school_id)       AS schools_monitored FROM monitoring_visits;
SELECT SUM(meals_served)               AS total_meals FROM monitoring_visits;
SELECT AVG(attendance_rate)            AS avg_attendance FROM monitoring_visits;
SELECT MIN(attendance_rate), MAX(attendance_rate) FROM monitoring_visits;
SELECT STDDEV_SAMP(attendance_rate), VAR_SAMP(attendance_rate) FROM monitoring_visits;
-- PostgreSQL: STDDEV(...) and VARIANCE(...) are aliases for the sample versions by default.
 
-- GROUP BY — one summary row per group
SELECT report_month,
       COUNT(*)                    AS visits,
       ROUND(AVG(attendance_rate),1) AS avg_attendance,
       SUM(meals_served)            AS total_meals
FROM   monitoring_visits
GROUP BY report_month
ORDER BY report_month;
 
-- Grouping by a joined dimension
SELECT s.school_type,
       s.ownership,
       COUNT(*)                    AS visits,
       ROUND(AVG(mv.dropout_rate),2) AS avg_dropout_rate
FROM   monitoring_visits mv
JOIN   schools s ON s.school_id = mv.school_id
GROUP BY s.school_type, s.ownership
ORDER BY avg_dropout_rate DESC;
 
-- HAVING — filters groups AFTER aggregation (WHERE cannot reference aggregates)
SELECT school_id, COUNT(*) AS months_with_stockouts
FROM   monitoring_visits
WHERE  stock_out = 'yes'
GROUP BY school_id
HAVING COUNT(*) >= 3
ORDER BY months_with_stockouts DESC;
 
-- WITH ROLLUP — MySQL's version of subtotal/grand-total aggregation
SELECT s.ownership, s.school_type, COUNT(*) AS visits
FROM   monitoring_visits mv JOIN schools s ON s.school_id = mv.school_id
GROUP BY s.ownership, s.school_type WITH ROLLUP;
-- Note: MySQL has no GROUPING SETS or CUBE syntax — WITH ROLLUP only builds
-- the hierarchical subtotal path (ownership, then ownership+school_type, then grand total).
-- PostgreSQL: GROUP BY ROLLUP (s.ownership, s.school_type);   -- or GROUPING SETS(...) / CUBE(...)
 
-- Conditional aggregation (MySQL has no FILTER clause — use CASE inside the aggregate)
SELECT report_month,
       SUM(CASE WHEN stock_out = 'yes' THEN 1 ELSE 0 END)          AS stockout_visits,
       SUM(CASE WHEN complaints_received = 'yes' THEN 1 ELSE 0 END) AS complaint_visits
FROM   monitoring_visits
GROUP BY report_month
ORDER BY report_month;
-- PostgreSQL: COUNT(*) FILTER (WHERE stock_out = 'yes')   -- native, more concise

Joins

Joins combine rows from related tables. INNER keeps only matches, LEFT/RIGHT keep unmatched rows from one side, FULL keeps both, CROSS produces every combination, and SELF joins a table to itself.

Joins 

/* ----------------------------------------------------------------------------
  5.  JOINS
---------------------------------------------------------------------------- */
 
-- INNER JOIN — only rows with a match in both tables
SELECT mv.visit_id, s.school_name, mv.report_month, mv.attendance_rate
FROM   monitoring_visits mv
INNER JOIN schools s ON s.school_id = mv.school_id;
 
-- LEFT (OUTER) JOIN — all schools, even those with zero recorded visits
SELECT s.school_id, s.school_name, COUNT(mv.visit_id) AS visit_count
FROM   schools s
LEFT JOIN monitoring_visits mv ON mv.school_id = s.school_id
GROUP BY s.school_id, s.school_name
ORDER BY visit_count ASC;                          -- surfaces under-monitored schools
 
-- RIGHT JOIN — symmetrical opposite of LEFT JOIN (less commonly used)
SELECT mv.visit_id, s.school_name
FROM   schools s
RIGHT JOIN monitoring_visits mv ON mv.school_id = s.school_id;
 
-- FULL OUTER JOIN — MySQL has NO native FULL OUTER JOIN; emulate with
-- LEFT JOIN UNION RIGHT JOIN (or UNION ALL + de-dup logic as needed)
SELECT s.school_id AS schools_tbl_id, mv.school_id AS visits_tbl_id
FROM   schools s
LEFT JOIN monitoring_visits mv ON mv.school_id = s.school_id
WHERE  mv.school_id IS NULL
UNION
SELECT s.school_id, mv.school_id
FROM   schools s
RIGHT JOIN monitoring_visits mv ON mv.school_id = s.school_id
WHERE  s.school_id IS NULL;
-- PostgreSQL: FULL OUTER JOIN is native —
-- SELECT s.school_id, mv.school_id FROM schools s FULL OUTER JOIN monitoring_visits mv
--   ON mv.school_id = s.school_id WHERE s.school_id IS NULL OR mv.school_id IS NULL;
 
-- CROSS JOIN — Cartesian product; useful for generating a complete
-- school-x-month scaffold to detect MISSING monthly reports
SELECT s.school_id, gm.report_month
FROM   schools s
CROSS JOIN (SELECT DISTINCT report_month FROM monitoring_visits) gm
WHERE  NOT EXISTS (
    SELECT 1 FROM monitoring_visits mv
    WHERE  mv.school_id = s.school_id AND mv.report_month = gm.report_month
);
 
-- SELF JOIN — compare a school's own consecutive-month performance
SELECT curr.school_id,
       curr.report_month              AS this_month,
       curr.attendance_rate            AS this_rate,
       prev.report_month               AS prior_month,
       prev.attendance_rate             AS prior_rate,
       curr.attendance_rate - prev.attendance_rate AS change
FROM   monitoring_visits curr
JOIN   monitoring_visits prev
       ON prev.school_id = curr.school_id
      AND prev.report_month = DATE_FORMAT(
              STR_TO_DATE(curr.report_month, '%Y-%m') - INTERVAL 1 MONTH, '%Y-%m');
 
-- Multi-table join (three levels of the geography hierarchy + facts)
SELECT r.region_name, c.county_name, s.school_name, mv.report_month, mv.attendance_rate
FROM   monitoring_visits mv
JOIN   schools s        ON s.school_id      = mv.school_id
JOIN   sub_counties sc  ON sc.sub_county_id = s.sub_county_id
JOIN   counties c       ON c.county_id      = sc.county_id
JOIN   regions r         ON r.region_id      = c.region_id
WHERE  r.region_name = 'Coast';

Subqueries

A subquery nests inside another statement — in WHERE, FROM, or SELECT. A correlated subquery references the outer query and re-runs per outer row.

Subqueries 

/* ----------------------------------------------------------------------------
  6.  SUBQUERIES
---------------------------------------------------------------------------- */
 
-- Scalar subquery in SELECT
SELECT school_id, report_month, attendance_rate,
       (SELECT AVG(attendance_rate) FROM monitoring_visits) AS overall_avg
FROM   monitoring_visits;
 
-- Subquery in WHERE with a scalar comparison
SELECT school_id, report_month, meals_served
FROM   monitoring_visits
WHERE  meals_served > (SELECT AVG(meals_served) FROM monitoring_visits);
 
-- IN subquery — schools that ever had a stockout
SELECT school_name FROM schools
WHERE  school_id IN (SELECT school_id FROM monitoring_visits WHERE stock_out = 'yes');
 
-- NOT IN subquery — schools that NEVER had a stockout
-- (guard with "IS NOT NULL" inside NOT IN — a single NULL poisons the whole result,
--  in MySQL exactly as in every other RDBMS)
SELECT school_name FROM schools
WHERE  school_id NOT IN (
    SELECT school_id FROM monitoring_visits WHERE stock_out = 'yes' AND school_id IS NOT NULL
);
 
-- EXISTS / NOT EXISTS — usually faster than IN for large sets, short-circuits per row
SELECT s.school_name
FROM   schools s
WHERE  EXISTS (
    SELECT 1 FROM monitoring_visits mv
    WHERE  mv.school_id = s.school_id AND mv.complaints_received = 'yes'
);
 
-- Correlated subquery — re-evaluated per outer row; here, each school's latest visit
SELECT s.school_id, s.school_name,
       (SELECT mv.attendance_rate
        FROM   monitoring_visits mv
        WHERE  mv.school_id = s.school_id
        ORDER BY mv.report_month DESC
        LIMIT 1)                                    AS latest_attendance_rate
FROM   schools s;
 
-- Subquery in FROM (a "derived table") — must be aliased
SELECT school_type, ROUND(AVG(monthly_meals),0) AS avg_meals_per_school_month
FROM (
    SELECT s.school_type, mv.school_id, mv.report_month, SUM(mv.meals_served) AS monthly_meals
    FROM   monitoring_visits mv JOIN schools s ON s.school_id = mv.school_id
    GROUP BY s.school_type, mv.school_id, mv.report_month
) AS monthly_totals
GROUP BY school_type;
 
-- ANY / ALL — supported identically in MySQL
SELECT school_id, meals_served FROM monitoring_visits
WHERE  meals_served > ALL (SELECT meals_served FROM monitoring_visits WHERE report_month = '2025-01');
 
SELECT school_id, meals_served FROM monitoring_visits
WHERE  meals_served > ANY (SELECT meals_served FROM monitoring_visits WHERE report_month = '2025-01');

Common Table Expressions

A CTE (WITH clause) defines a named, temporary result set the main query can reference. A recursive CTE references itself to walk hierarchical data.

Common Table Expressions 

/* ----------------------------------------------------------------------------
  7.  COMMON TABLE EXPRESSIONS (CTEs) — MySQL 8.0+
---------------------------------------------------------------------------- */
 
-- Basic CTE — improves readability vs. nested subqueries
WITH monthly_avg AS (
    SELECT report_month, AVG(attendance_rate) AS avg_rate
    FROM   monitoring_visits
    GROUP BY report_month
)
SELECT * FROM monthly_avg ORDER BY report_month;
 
-- Multiple CTEs, chained together
WITH school_meals AS (
    SELECT school_id, SUM(meals_served) AS total_meals
    FROM   monitoring_visits
    GROUP BY school_id
),
school_ranked AS (
    SELECT school_id, total_meals,
           RANK() OVER (ORDER BY total_meals DESC) AS meal_rank
    FROM   school_meals
)
SELECT sr.meal_rank, s.school_name, sr.total_meals
FROM   school_ranked sr
JOIN   schools s ON s.school_id = sr.school_id
WHERE  sr.meal_rank <= 10;
 
-- Recursive CTE — walk the region → county hierarchy (MySQL syntax is
-- WITH RECURSIVE, same as Postgres)
WITH RECURSIVE geo_tree AS (
    SELECT region_id AS node_id, region_name AS node_name, CAST(NULL AS SIGNED) AS parent_id, 1 AS depth
    FROM   regions
    UNION ALL
    SELECT c.county_id + 100000, c.county_name, gt.node_id, gt.depth + 1
    FROM   counties c
    JOIN   geo_tree gt ON gt.node_id = c.region_id AND gt.depth = 1
)
SELECT * FROM geo_tree ORDER BY depth, node_name;
 
-- CTE used purely to stage a filter, then reused twice
WITH low_attendance AS (
    SELECT * FROM monitoring_visits WHERE attendance_rate < 75
)
SELECT report_month, COUNT(*) AS low_attendance_visits FROM low_attendance GROUP BY report_month
UNION ALL
SELECT 'ALL-TIME', COUNT(*) FROM low_attendance;

Window Functions

Window functions compute a value across related rows (OVER()) without collapsing them. ROW_NUMBER, RANK, and LAG/LEAD support rankings and period comparisons.

Window Functions 

/* ----------------------------------------------------------------------------
  8.  WINDOW FUNCTIONS — MySQL 8.0+
---------------------------------------------------------------------------- */
 
-- ROW_NUMBER — unique sequential rank per partition
SELECT school_id, report_month, attendance_rate,
       ROW_NUMBER() OVER (PARTITION BY school_id ORDER BY report_month) AS visit_seq
FROM   monitoring_visits;
 
-- RANK / DENSE_RANK — schools ranked by total meals served (ties handled differently)
SELECT school_id, SUM(meals_served) AS total_meals,
       RANK()       OVER (ORDER BY SUM(meals_served) DESC) AS rnk,        -- leaves gaps after ties
       DENSE_RANK() OVER (ORDER BY SUM(meals_served) DESC) AS dense_rnk   -- no gaps
FROM   monitoring_visits
GROUP BY school_id;
 
-- LAG / LEAD — compare each month to the previous/next one, per school
SELECT school_id, report_month, attendance_rate,
       LAG(attendance_rate)  OVER (PARTITION BY school_id ORDER BY report_month) AS prev_month_rate,
       LEAD(attendance_rate) OVER (PARTITION BY school_id ORDER BY report_month) AS next_month_rate,
       attendance_rate - LAG(attendance_rate) OVER (PARTITION BY school_id ORDER BY report_month) AS mom_change
FROM   monitoring_visits;
 
-- Running totals & moving averages
SELECT school_id, report_month, meals_served,
       SUM(meals_served) OVER (PARTITION BY school_id ORDER BY report_month
                                 ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_total_meals,
       AVG(attendance_rate) OVER (PARTITION BY school_id ORDER BY report_month
                                    ROWS BETWEEN 2 PRECEDING AND CURRENT ROW) AS rolling_3mo_avg_attendance
FROM   monitoring_visits;
 
-- NTILE — bucket schools into quartiles by average attendance
SELECT school_id, avg_attendance,
       NTILE(4) OVER (ORDER BY avg_attendance DESC) AS attendance_quartile
FROM  (SELECT school_id, AVG(attendance_rate) AS avg_attendance
       FROM monitoring_visits GROUP BY school_id) t;
 
-- FIRST_VALUE / LAST_VALUE — each school's first vs latest recorded attendance rate
SELECT DISTINCT school_id,
       FIRST_VALUE(attendance_rate) OVER w AS first_rate,
       LAST_VALUE(attendance_rate)  OVER (PARTITION BY school_id ORDER BY report_month
                                            ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) AS latest_rate
FROM   monitoring_visits
WINDOW w AS (PARTITION BY school_id ORDER BY report_month
             ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING);
 
-- PERCENT_RANK / CUME_DIST — relative standing within the whole dataset
SELECT school_id, report_month, attendance_rate,
       ROUND(PERCENT_RANK() OVER (ORDER BY attendance_rate), 3) AS pct_rank,
       ROUND(CUME_DIST()    OVER (ORDER BY attendance_rate), 3) AS cume_dist
FROM   monitoring_visits;

Set Operations

UNION merges and dedupes two result sets, UNION ALL keeps duplicates, INTERSECT returns common rows, and EXCEPT (MINUS in Oracle) returns rows only in the first query.

Set Operations 

/* ----------------------------------------------------------------------------
  9.  SET OPERATIONS
---------------------------------------------------------------------------- */
 
-- UNION — combine two result sets, removing duplicates; column count/types must line up
SELECT school_id FROM monitoring_visits WHERE stock_out = 'yes'
UNION
SELECT school_id FROM monitoring_visits WHERE complaints_received = 'yes';
 
-- UNION ALL — same but keeps duplicates (faster — no de-dup step)
SELECT school_id, 'stockout' AS flag FROM monitoring_visits WHERE stock_out = 'yes'
UNION ALL
SELECT school_id, 'complaint' AS flag FROM monitoring_visits WHERE complaints_received = 'yes';
 
-- INTERSECT — rows present in BOTH result sets (MySQL 8.0.31+ only)
SELECT school_id FROM monitoring_visits WHERE stock_out = 'yes'
INTERSECT
SELECT school_id FROM monitoring_visits WHERE complaints_received = 'yes';
 
-- EXCEPT — rows in the first set but NOT the second (MySQL 8.0.31+ only)
SELECT DISTINCT school_id FROM schools
EXCEPT
SELECT DISTINCT school_id FROM monitoring_visits;      -- schools never visited at all
 
-- On older MySQL (pre-8.0.31), emulate INTERSECT / EXCEPT with IN / NOT IN or JOIN:
SELECT DISTINCT a.school_id                              -- INTERSECT emulation
FROM   monitoring_visits a
WHERE  a.stock_out = 'yes'
  AND  a.school_id IN (SELECT school_id FROM monitoring_visits WHERE complaints_received = 'yes');
 
SELECT DISTINCT s.school_id                               -- EXCEPT emulation
FROM   schools s
WHERE  s.school_id NOT IN (SELECT school_id FROM monitoring_visits WHERE school_id IS NOT NULL);
-- PostgreSQL: both INTERSECT and EXCEPT are native from every supported version.

Conditional Logic

CASE WHEN evaluates conditions in order and returns the first match, functioning like an inline if/else inside a query.

Conditional Logic 

/* ----------------------------------------------------------------------------
  10.  CONDITIONAL LOGIC
---------------------------------------------------------------------------- */
 
-- Simple CASE (value match)
SELECT school_id, ownership,
       CASE ownership
            WHEN 'Public'  THEN 'Government-funded'
            WHEN 'Private' THEN 'Privately-funded'
            ELSE 'Unknown'
       END AS funding_label
FROM   schools;
 
-- Searched CASE (boolean conditions) — building a performance category
SELECT school_id, report_month, attendance_rate,
       CASE
            WHEN attendance_rate >= 90 THEN 'Excellent'
            WHEN attendance_rate >= 75 THEN 'Satisfactory'
            WHEN attendance_rate >= 50 THEN 'At risk'
            ELSE 'Critical'
       END AS attendance_band
FROM   monitoring_visits;
 
-- CASE inside an aggregate (conditional counting / "pivoting")
SELECT report_month,
       SUM(CASE WHEN meal_type = 'lunch' THEN 1 ELSE 0 END)  AS lunch_only_visits,
       SUM(CASE WHEN meal_type = 'both'  THEN 1 ELSE 0 END)  AS both_meals_visits
FROM   monitoring_visits
GROUP BY report_month;
 
-- IFNULL / COALESCE — first non-NULL value
SELECT school_id, IFNULL(meal_type, 'not recorded') AS meal_type_clean FROM monitoring_visits;
SELECT school_id, COALESCE(meal_type, 'not recorded') AS meal_type_clean FROM monitoring_visits;  -- also works
-- PostgreSQL: no IFNULL — use COALESCE(meal_type, 'not recorded') instead.
-- SQL Server:  ISNULL(meal_type, 'not recorded')
 
-- NULLIF — returns NULL if two values are equal (prevents divide-by-zero, used throughout above)
SELECT school_id, meals_served, attendance_today,
       meals_served / NULLIF(attendance_today, 0) AS meals_per_attendee
FROM   monitoring_visits;
 
-- GREATEST / LEAST — identical in MySQL and PostgreSQL
SELECT school_id, GREATEST(dropouts_boys, dropouts_girls) AS higher_dropout_group
FROM   monitoring_visits;
 
-- IF() — MySQL-specific two-way shorthand (no equivalent keyword in Postgres)
SELECT school_id, IF(attendance_rate >= 80, 'On track', 'Needs support') AS status
FROM   monitoring_visits;
-- PostgreSQL: use CASE WHEN ... THEN ... ELSE ... END instead — no bare IF() in plain SQL.

Transaction Control & Data Control Language

This tab covers TCL (Transaction Control Language), which groups statements into atomic units of work, and DCL (Data Control Language), which manages who is allowed to do what.

Transactions

A transaction bundles statements into one atomic unit (ACID). BEGIN starts it, COMMIT makes it permanent, ROLLBACK undoes it, and SAVEPOINT marks a point a rollback can return to.

Transactions 

/* ############################################################################
  TCL (TRANSACTION CONTROL LANGUAGE)
############################################################################ */
 
-- A transaction groups multiple statements so they succeed or fail TOGETHER
-- (atomicity) — critical when several related tables must stay consistent,
-- e.g. recording a visit AND updating the school's tier in one go.
-- Requires InnoDB (MyISAM does not support transactions).
 
START TRANSACTION;                         -- PostgreSQL: BEGIN;   (both work in MySQL too)
 
    UPDATE schools
    SET    feeding_program_tier = 'Tier 2'
    WHERE  school_id = 'SCH0001';
 
    INSERT INTO monitoring_visits (school_id, report_month, endtime, enrolment_total,
                                    enrolment_boys, enrolment_girls, attendance_today, attendance_rate)
    VALUES ('SCH0001', '2026-05', NOW(), 500, 250, 250, 470, 94.0);
 
COMMIT;                                    -- persist both changes permanently
 
-- If something goes wrong mid-transaction, roll everything back:
START TRANSACTION;
    UPDATE monitoring_visits SET meals_served = meals_served + 100 WHERE school_id = 'SCH0001';
    -- ... suppose a validation step fails here ...
ROLLBACK;                                   -- undoes the UPDATE entirely; nothing is saved
 
-- SAVEPOINT — partial rollback within a larger transaction
START TRANSACTION;
    UPDATE schools SET is_active = 0 WHERE school_id = 'SCH0001';
 
    SAVEPOINT before_visit_delete;
    DELETE FROM monitoring_visits WHERE school_id = 'SCH0001' AND report_month = '2025-01';
 
    ROLLBACK TO SAVEPOINT before_visit_delete;  -- undo only the DELETE, keep the school update
COMMIT;
 
-- Transaction isolation levels (control how concurrent transactions see each other's changes)
SET SESSION TRANSACTION ISOLATION LEVEL READ COMMITTED;
-- other levels: REPEATABLE READ (MySQL/InnoDB default), READ UNCOMMITTED, SERIALIZABLE
-- PostgreSQL: BEGIN TRANSACTION ISOLATION LEVEL READ COMMITTED;  (Postgres default is also READ COMMITTED)
 
-- Locking a row explicitly for update (prevents two enumerators editing the same visit at once)
START TRANSACTION;
    SELECT * FROM monitoring_visits WHERE visit_id = 1 FOR UPDATE;
    UPDATE monitoring_visits SET reviewed_by = 'Maurice Juma' WHERE visit_id = 1;
COMMIT;
 
-- Autocommit note: MySQL runs with autocommit = ON by default — every
-- standalone statement is its own implicit transaction unless you explicitly
-- open one with START TRANSACTION / BEGIN.
SET autocommit = 0;    -- disable, to keep a manual multi-statement transaction open
SET autocommit = 1;    -- restore the default

User and Permission Management

CREATE USER provisions a login, GRANT gives privileges on an object, and REVOKE removes them. Grouping privileges into roles keeps large permission sets manageable.

User and Permission Management 

/* ############################################################################
  DCL (USER AND PERMISSION MANAGEMENT)
############################################################################ */
 
-- MySQL identifies accounts as 'user'@'host' — the host part matters:
-- '%' = allow connecting from any host, 'localhost' = only from the server itself.
 
-- Create users
CREATE USER 'meal_admin'@'%'      IDENTIFIED BY 'Str0ngPass!';
CREATE USER 'meal_analyst'@'%'    IDENTIFIED BY 'Str0ngPass!';
CREATE USER 'meal_enumerator'@'%' IDENTIFIED BY 'Str0ngPass!';
 
-- PostgreSQL: CREATE ROLE meal_admin WITH LOGIN PASSWORD 'Str0ngPass!' CREATEDB;
 
-- MySQL 8.0+ also supports true ROLE objects, separate from login users —
-- useful for the "grant privileges to a role, then hand the role to users" pattern:
CREATE ROLE 'meal_readonly';
 
-- Grant broad privileges to the admin
GRANT ALL PRIVILEGES ON school_feeding_db.* TO 'meal_admin'@'%';
 
-- Analysts: read-only on everything
GRANT SELECT ON school_feeding_db.* TO 'meal_analyst'@'%';
GRANT SELECT ON school_feeding_db.mv_region_summary TO 'meal_analyst'@'%';
 
-- Enumerators: can only insert new visits and update specific columns —
-- no DELETE, no access to user-management tables
GRANT SELECT, INSERT ON school_feeding_db.monitoring_visits TO 'meal_enumerator'@'%';
GRANT UPDATE (attendance_today, attendance_boys, attendance_girls, meals_served, stock_level)
      ON school_feeding_db.monitoring_visits TO 'meal_enumerator'@'%';    -- column-level GRANT
GRANT SELECT ON school_feeding_db.schools TO 'meal_enumerator'@'%';
GRANT SELECT ON school_feeding_db.sub_counties TO 'meal_enumerator'@'%';
GRANT SELECT ON school_feeding_db.counties TO 'meal_enumerator'@'%';
GRANT SELECT ON school_feeding_db.regions TO 'meal_enumerator'@'%';
 
-- Group role pattern: grant privileges to the role, then hand the role to users
GRANT SELECT ON school_feeding_db.* TO 'meal_readonly';
GRANT 'meal_readonly' TO 'meal_analyst'@'%';
SET DEFAULT ROLE 'meal_readonly' TO 'meal_analyst'@'%';    -- so the role is active on login
 
-- Revoke privileges
REVOKE INSERT ON school_feeding_db.monitoring_visits FROM 'meal_analyst'@'%';   -- analysts should never write
REVOKE ALL PRIVILEGES, GRANT OPTION FROM 'meal_enumerator'@'%' ON school_feeding_db.schools_archive;
 
-- View who can access what
SHOW GRANTS FOR 'meal_analyst'@'%';
SELECT grantee, table_name, privilege_type
FROM   information_schema.table_privileges
WHERE  table_schema = 'school_feeding_db'
ORDER BY grantee, table_name;
-- PostgreSQL: SELECT grantee, table_name, privilege_type FROM information_schema.role_table_grants
--             WHERE table_schema = 'meal_program' ORDER BY grantee, table_name;
 
-- Alter a user: change password, rename, set expiry
ALTER USER 'meal_enumerator'@'%' IDENTIFIED BY 'NewStr0ngerPass!';
ALTER USER 'meal_enumerator'@'%' PASSWORD EXPIRE INTERVAL 180 DAY;
RENAME USER 'meal_analyst'@'%' TO 'meal_reporting_analyst'@'%';
 
-- Drop a user / role
DROP USER IF EXISTS 'meal_readonly';
-- (dropping a role also removes it from any users it was granted to)
 
-- Apply changes immediately when privilege tables were edited directly
-- (not usually needed after plain GRANT/REVOKE statements, but good practice to know)
FLUSH PRIVILEGES;
